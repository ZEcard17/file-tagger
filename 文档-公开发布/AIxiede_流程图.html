8. 数据库数据结构

erDiagram
    files ||--o{ file_tags : "文件-标签"
    tags ||--o{ file_tags : "标签-文件"
    tags ||--o{ tag_relations : "父子关系"
    files ||--o{ category_files : "分类-文件"
    categories ||--o{ category_files : "分类成员"
    categories ||--o{ category_tag_links : "分类-标签超链接"
    tags ||--o{ category_tag_links : "超链接标签"
    
    files {
        int id PK
        string path
        string folder
        int size
        string mtime
    }
    tags {
        int id PK
        string name
        string color
        int sort_order
    }
    file_tags {
        int file_id FK
        int tag_id FK
        int is_auto
    }
    tag_relations {
        int parent_id FK
        int child_id FK
    }
    categories {
        int id PK
        string name
        string color
        string icon
    }
    category_files {
        int category_id FK
        int file_id FK
    }
    category_tag_links {
        int category_id FK
        int tag_id FK
    }
